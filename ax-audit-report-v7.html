🔒 Pillar 2: Validated Domain Primitives

Pattern: @JvmInline value class with private constructors, validated factories returning DomainResult<T>

Example: NonBlankId.kt 
@JvmInline
value class NonBlankId private constructor(val value: String) {
    companion object {
        fun create(raw: String?, fieldName: String): DomainResult<NonBlankId> {
            if (raw.isNullOrBlank()) {
                return DomainResult.Failure(
                    ProblemDetails(
                        type = TYPE_VALIDATION,
                        detail = "$fieldName must not be blank",
                        suggestedAction = "Provide a non-blank identifier",
                        isRetriable = false
                    )
                )
            }
            return DomainResult.Success(NonBlankId(raw))
        }
    }
}
